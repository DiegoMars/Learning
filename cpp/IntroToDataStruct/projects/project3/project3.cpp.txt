// A string may use more than one type of delimiter to bracket information into "blocks."
// For example, A string may use braces { }, parentheses ( ), and brackets [ ] as delimiters.
// A string is properly delimited if each right delimiter is matched with a preceding left
// delimiter of the same type in such a way that either the resulting blocks of information
// are disjoint, or one of them is completely nested within the other. Write a program
// that uses a single stack to check whether a string containing braces, parentheses, and
// brackets is properly delimited.
#include <iostream>
#include <string>
#include "derivedLinkedStack.h"

using namespace std;

int main()
{
    // So, my thought process for this one is to go through the string, and if there is
    // an instance of a left delimiter, add it to the stack. If there is an instance of
    // a right delimitor, compare it to the top of the stack. If it matches, pop the top
    // of the stack. If it doesn't match or the stack it empty, end the program and print
    // an error code. If it reaches the end of the string no problem, end program and
    // print that it is properly delimited
    linkedStackType<char> stack;
    stack.initializeList(); // Initialize empty stack
    string input;
    char lDelimiters[3] = {'{', '(', '['};
    char rDelimiters[3] = {'}', ')', ']'};
    bool flag = true;

    cout << "Enter string to check if it is properly delimited: ";
    getline(cin, input);

    int i = 0;
    while (flag && i < input.length())
    {
        char c = input[i]; // Lil Simplification

        // if (c == '{' || c == '(' || c == '[')
        // {
        //     stack.push(c);
        // }
        // if (c == '}' || c == ')' || c == ']')
        // {
        //     if (stack.isEmptyStack() || stack.top())
        // }
        // At first I used if statements, but I lowkey think it would be easier to use loops

        int j = 0;
        while (flag && j < 3)
        {
            char left = lDelimiters[j];
            char right = rDelimiters[j]; // Lil more simplification

            if (c == left)
            {
                stack.push(c);
            }
            else if (c == right)
            {
                // If the stack is empty or the delimiters don't match, say string is not
                // properly delimited and end loops. Otherwise, pop the top of the stack 
                if (stack.isEmptyStack() || stack.top() != left){
                    cout << "String is not properly delimited!!\n" << endl;
                    flag = false;
                } else {
                    stack.pop();
                }
            }
            j++;
        }
        i++;
    }
    if (flag){
        cout << "String is properly delimited\n" << endl;
    }

    return 0;
}
